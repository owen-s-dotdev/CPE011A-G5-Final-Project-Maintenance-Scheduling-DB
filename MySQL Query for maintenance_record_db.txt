
-- Create the database
CREATE DATABASE IF NOT EXISTS maintenance_record_db;
USE maintenance_record_db;

-- 1. Create Departments Table
CREATE TABLE IF NOT EXISTS Departments (
    DepartmentID INT AUTO_INCREMENT PRIMARY KEY,
    DepartmentName VARCHAR(100) NOT NULL,
    Location VARCHAR(100)
);

-- 2. Create Technician Table
CREATE TABLE IF NOT EXISTS Technician (
    TechnicianID INT AUTO_INCREMENT PRIMARY KEY,
    FirstName VARCHAR(50) NOT NULL,
    LastName VARCHAR(50) NOT NULL,
    ContactNumber VARCHAR(20),
    Email VARCHAR(100)
);

-- 3. Create DeviceTypes Table
CREATE TABLE IF NOT EXISTS DeviceTypes (
    DeviceTypeID INT AUTO_INCREMENT PRIMARY KEY,
    TypeName VARCHAR(100) NOT NULL,
    DepartmentID INT,
    FOREIGN KEY (DepartmentID) REFERENCES Departments(DepartmentID) ON DELETE SET NULL
);

-- 4. Create MaintenanceRecord Table
CREATE TABLE IF NOT EXISTS MaintenanceRecord (
    RecordID INT AUTO_INCREMENT PRIMARY KEY,
    TechnicianID INT,
    CompletionDate DATE,
    Notes TEXT,
    FOREIGN KEY (TechnicianID) REFERENCES Technician(TechnicianID) ON DELETE SET NULL
);

-- 5. Create Devices Table
CREATE TABLE IF NOT EXISTS Devices (
    DeviceID INT AUTO_INCREMENT PRIMARY KEY,
    RecordID INT,
    DeviceName VARCHAR(100) NOT NULL,
    DeviceTypeID INT,
    SerialNumber VARCHAR(100),
    PurchaseDate DATE,
    Status VARCHAR(50),
    FOREIGN KEY (RecordID) REFERENCES MaintenanceRecord(RecordID) ON DELETE SET NULL,
    FOREIGN KEY (DeviceTypeID) REFERENCES DeviceTypes(DeviceTypeID) ON DELETE SET NULL
);
